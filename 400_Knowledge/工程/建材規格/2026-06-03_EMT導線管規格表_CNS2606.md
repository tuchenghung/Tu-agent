## 四、選管邏輯（待更新程式碼）

```python
import math

# CNS 2606 C4060 實際內徑（mm）
EMT_INNER_DIAMETER = {
    'E-19': 16.7,
    'E-25': 23.0,
    'E-31': 29.0,
    'E-39': 35.3,
    'E-51': 48.0,
    'E-63': 60.3,
    'E-75': 72.6,
}

FILL_RATIO = 0.40  # CNS 規定 40%

def get_emt_fill_area(size_code):
    d = EMT_INNER_DIAMETER[size_code]
    total_area = math.pi / 4 * d ** 2
    return total_area * FILL_RATIO

def select_emt(wire_total_area_mm2):
    """
    依導線總截面積（含絕緣皮）選 EMT 管
    wire_total_area_mm2: 管內所有導線截面積加總
    """
    for code, inner_d in EMT_INNER_DIAMETER.items():
        fill_area = math.pi / 4 * inner_d ** 2 * FILL_RATIO
        if wire_total_area_mm2 <= fill_area:
            return code, inner_d
    return 'E-75+', None  # 超出最大管徑

# 範例：3條 38mm² 導線（含接地）
# 每條截面積（含絕緣）約 65mm²，共 3 條 = 195mm²
# select_emt(195) → E-51（允許 723.8mm²）
```
